```cisco
enable
configure terminal
ip access-list standard MGMT_LA_LIBERTAD
 permit 10.192.44.208 0.0.0.15
 permit 10.192.42.176 0.0.0.15
deny any
exit
line vty 0 4
 access-class MGMT_LA_LIBERTAD in
 transport input ssh
 login local
 exec-timeout 10 0
exit
end
```

```cisco
ip access-list extended FTP_LL

end
```
