# 3.6.5 Sede sucursal 4 - Puno

## Primera politica de seguridad

```cisco
enable
configure terminal
ip access-list standard MGMT_PUNO
 permit 10.192.56.128 0.0.0.15
 permit 10.192.42.176 0.0.0.15
deny any
exit
line vty 0 4
 access-class MGMT_PUNO in
 transport input ssh
 login local
 exec-timeout 10 0
exit
end
```

## Segunda politica de seguridad

```cisco
enable
configure terminal
ip access-list extended FTP_PUNO
 permit tcp 10.192.56.0 0.0.3.255 host 10.192.56.197 eq 21
 permit tcp 10.192.40.0 0.0.3.255 host 10.192.56.197 eq 21
 deny tcp any host 10.192.56.197 eq 21
 permit ip any any
exit
interface GigabitEthernet0/0.58
 ip access-group FTP_PUNO in
exit
end
```
